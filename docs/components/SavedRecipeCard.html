import { Component, input, output } from '@angular/core'
import { RouterLink } from '@angular/router'
import { Post } from '@core/models/post/post.model'

/**
 * Componente que representa una tarjeta visual simplificada para mostrar una receta o post guardado en la biblioteca.
 */
@Component({
	selector: 'app-saved-recipe-card',
	imports: [RouterLink],
	templateUrl: './saved-recipe-card.html',
})
export class SavedRecipeCard {
	/**
	 * La publicación guardada que se va a renderizar.
	 */
	post = input.required<Post>()

	/**
	 * Emisor de evento que se dispara al solicitar quitar la publicación de la lista de guardados.
	 */
	remove = output<string>()
}

<article class="group overflow-hidden rounded-[1.75rem] border border-outline bg-surface-container shadow-1 transition-all duration-fast ease-standard hover:-translate-y-0.5 hover:shadow-3">
	<div class="relative aspect-[4/3] overflow-hidden bg-secondary-container">
		@if (post().media[0]) {
			<img [src]="post().media[0].url" [alt]="post().title ?? ''" class="size-full object-cover transition duration-500 group-hover:scale-105" />
		} @else {
			<div class="grid size-full place-items-center text-secondary">
				<i class="pi pi-image text-5xl"></i>
			</div>
		}
		<div class="absolute inset-x-0 top-0 flex items-start justify-between p-4">
			@if (post().categories[0]) {
				<span class="rounded-pill bg-black/65 px-3 py-1 text-label-md font-bold text-white">{{ post().categories[0] }}</span>
			} @else {
				<span></span>
			}
			<button
				type="button"
				class="grid size-11 place-items-center rounded-pill bg-surface text-primary shadow-2 transition-colors duration-fast ease-standard hover:bg-primary hover:text-on-primary active:scale-[0.98]"
				(click)="remove.emit(post().id)"
				aria-label="Quitar guardado"
			>
				<i class="pi pi-bookmark-fill"></i>
			</button>
		</div>
		@if (post().recipe?.timeRequired) {
			<span class="absolute bottom-4 right-4 rounded-pill bg-surface/90 px-3 py-1 text-label-md font-bold text-on-surface">{{ post().recipe!.timeRequired }} min</span>
		}
	</div>

	<div class="p-4 sm:p-5">
		<div class="flex items-center gap-2">
			@if (post().author.photoUrl) {
				<img [src]="post().author.photoUrl" [alt]="post().author.name ?? ''" class="size-9 rounded-pill object-cover" />
			} @else {
				<span class="grid size-9 shrink-0 place-items-center rounded-pill bg-primary-container text-label-md font-bold text-on-primary-container">
					{{ (post().author.name ?? post().author.username ?? '?').charAt(0).toUpperCase() }}
				</span>
			}
			<span class="truncate text-body-sm font-bold text-on-surface-muted">{{ post().author.name ?? post().author.username }}</span>
		</div>

		<h3 class="mt-4 line-clamp-2 text-title-md font-bold leading-snug text-on-surface">
			{{ post().title ?? post().recipe?.name ?? post().description }}
		</h3>
		<p class="mt-2 line-clamp-2 text-body-sm font-semibold leading-base text-on-surface-muted">{{ post().description }}</p>

		<div class="mt-5 flex items-center justify-between gap-3">
			<div class="flex min-w-0 items-center gap-3 text-body-sm font-bold text-on-surface-muted sm:gap-4">
				<span><i [class]="post().liked ? 'pi pi-heart-fill text-error' : 'pi pi-heart text-on-surface-muted'"></i> {{ post().likesCount }}</span>
				<span><i class="pi pi-comment"></i> {{ post().commentsCount }}</span>
				@if (post().recipe?.servings) {
					<span><i class="pi pi-users"></i> {{ post().recipe!.servings }}</span>
				}
			</div>
			<a [routerLink]="['/post', post().id]" class="inline-flex min-h-11 shrink-0 items-center gap-2 rounded-pill px-3 text-body-sm font-bold text-primary transition-colors duration-fast ease-standard hover:bg-primary-container">
				Ver
				<i class="pi pi-chevron-right text-xs"></i>
			</a>
		</div>
	</div>
</article>
