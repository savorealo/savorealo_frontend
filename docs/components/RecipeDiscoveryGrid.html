import { Component, input, output } from '@angular/core'
import { Post } from '@core/models/post/post.model'
import { RecipeDiscoveryCard } from '../recipe-discovery-card/recipe-discovery-card'
import { SavoLoader } from '@shared/components/savo-loader/savo-loader'

import { TranslatePipe } from '@shared/pipes/translate.pipe'

/**
 * Clase de utilidad para recipediscoverygrid.
 */
@Component({
	selector: 'app-recipe-discovery-grid',
	imports: [RecipeDiscoveryCard, SavoLoader, TranslatePipe],
	templateUrl: './recipe-discovery-grid.html',
})
export class RecipeDiscoveryGrid {
	/**
	 * Propiedad para gestionar posts.
	 */
	posts = input<Post[]>([])
	/**
	 * Propiedad para gestionar cargando.
	 */
	loading = input(false)
	/**
	 * Propiedad para gestionar cargando more.
	 */
	loadingMore = input(false)
	/**
	 * Indicador booleano para es o está empty.
	 */
	isEmpty = input(false)
	/**
	 * Propiedad para gestionar error.
	 */
	error = input<string | null>(null)

	/**
	 * Propiedad para gestionar retry.
	 */
	retry = output<void>()
	/**
	 * Propiedad para gestionar cargar more.
	 */
	loadMore = output<void>()
}

<section class="grid gap-5">
	@if (error()) {
		<div class="grid min-h-60 place-items-center rounded-2xl border border-error/30 bg-error-container p-8 text-center text-on-error-container">
			<div>
				<i class="pi pi-exclamation-triangle text-3xl"></i>
				<p class="mt-3 text-body-md font-bold">{{ error() }}</p>
				<button
					type="button"
					class="mt-4 inline-flex h-11 items-center gap-2 rounded-pill bg-on-error-container px-5 text-body-sm font-bold text-error-container transition-transform duration-fast ease-standard hover:shadow-3 active:scale-[0.98]"
					(click)="retry.emit()"
				>
					<i class="pi pi-refresh text-xs"></i>
					{{ 'explore.retry' | translate }}
				</button>
			</div>
		</div>
	} @else if (loading() && posts().length === 0) {
		<app-savo-loader [message]="'explore.searching_recipes' | translate" />
	} @else if (isEmpty()) {
		<div class="grid min-h-60 place-items-center rounded-2xl border border-outline bg-surface-container text-center text-on-surface-muted">
			<div>
				<i class="pi pi-search text-3xl"></i>
				<p class="mt-3 font-bold text-on-surface">{{ 'explore.no_recipes_category' | translate }}</p>
			</div>
		</div>
	} @else {
		<div class="grid gap-4 md:grid-cols-2 xl:grid-cols-3">
			@for (post of posts(); track post.id) {
				<app-recipe-discovery-card [post]="post" />
			}
		</div>
	}

	@if (loadingMore()) {
		<div class="h-14 animate-pulse rounded-2xl bg-surface-container"></div>
	}

	@if (!loading() && !isEmpty() && !error()) {
		<button
			type="button"
			class="inline-flex h-14 items-center justify-center gap-3 rounded-2xl border border-outline bg-surface-container text-body-md font-bold text-on-surface transition-colors duration-fast ease-standard hover:bg-surface-container-high"
			[disabled]="loadingMore()"
			(click)="loadMore.emit()"
		>
			<i class="pi pi-chevron-down text-sm"></i>
			<span>{{ loadingMore() ? ('explore.loading' | translate) : ('explore.load_more_recipes' | translate) }}</span>
		</button>
	}
</section>
