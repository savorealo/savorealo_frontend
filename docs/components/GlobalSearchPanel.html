import {
	Component,
	computed,
	effect,
	ElementRef,
	inject,
	OnDestroy,
	OnInit,
	PLATFORM_ID,
	viewChild,
} from '@angular/core'
import { isPlatformBrowser, NgClass } from '@angular/common'
import { GlobalSearchStore } from '@core/store/global-search.store'
import { TranslationService } from '@core/services/translation.service'
import { TranslatePipe } from '@shared/pipes/translate.pipe'

/**
 * Componente que representa el panel flotante de búsqueda global en la aplicación.
 * Permite buscar de forma interactiva y rápida recetas y usuarios mediante el atajo Ctrl+K / Cmd+K.
 */
@Component({
	selector: 'app-global-search-panel',
	templateUrl: './global-search-panel.html',
	styleUrls: ['./global-search-panel.css'],
	imports: [NgClass, TranslatePipe],
})
export class GlobalSearchPanel implements OnInit, OnDestroy {
	/**
	 * Almacén de estado que gestiona las consultas y resultados de la búsqueda global.
	 */
	readonly store = inject(GlobalSearchStore)

	/**
	 * Servicio para la traducción de etiquetas literales y de accesibilidad.
	 */
	readonly t = inject(TranslationService)

	/**
	 * Determina si la aplicación se ejecuta en el navegador (para poder añadir listeners de teclado).
	 */
	private readonly isBrowser = isPlatformBrowser(inject(PLATFORM_ID))

	/**
	 * Referencia de vista al elemento input HTML para la auto-focalización dinámica.
	 */
	private readonly inputEl = viewChild<ElementRef<HTMLInputElement>>('searchInput')

	/**
	 * Señal calculada que indica si se obtuvieron resultados de usuarios o publicaciones con la consulta actual.
	 */
	readonly hasResults = computed(() =>
		this.store.users().length > 0 || this.store.posts().length > 0,
	)

	/**
	 * Señal calculada que indica si la sección de usuarios debe renderizarse.
	 */
	readonly showUsers = computed(() =>
		this.store.tab() !== 'posts' && this.store.users().length > 0,
	)

	/**
	 * Señal calculada que indica si la sección de publicaciones debe renderizarse.
	 */
	readonly showPosts = computed(() =>
		this.store.tab() !== 'users' && this.store.posts().length > 0,
	)

	/**
	 * Crea una instancia de GlobalSearchPanel y establece un efecto para enfocar automáticamente el campo de texto cuando se abre el panel.
	 */
	constructor() {
		// Auto-focus input when panel opens
		effect(() => {
			if (this.store.open()) {
				setTimeout(() => this.inputEl()?.nativeElement.focus(), 50)
			}
		})
	}

	/**
	 * Método del ciclo de vida que agrega el listener de teclado global.
	 */
	ngOnInit(): void {
		if (this.isBrowser) document.addEventListener('keydown', this.handleKeydown)
	}

	/**
	 * Método del ciclo de vida para remover listeners de teclado al destruir el componente.
	 */
	ngOnDestroy(): void {
		if (this.isBrowser) document.removeEventListener('keydown', this.handleKeydown)
	}

	/**
	 * Manejador del evento keydown del documento global para capturar combinaciones del atajo de teclado de búsqueda y cierre por escape.
	 */
	private readonly handleKeydown = (e: KeyboardEvent): void => {
		// ⌘K or Ctrl+K → open
		if ((e.metaKey || e.ctrlKey) && e.key === 'k') {
			e.preventDefault()
			if (this.store.open()) {
				this.store.closePanel()
			} else {
				this.store.openPanel()
			}
			return
		}
		// Escape → close
		if (e.key === 'Escape' && this.store.open()) {
			this.store.closePanel()
		}
	}

	/**
	 * Evento al teclear en el input de búsqueda. Modifica la query en el almacén de búsqueda global.
	 * @param event Evento de entrada de caracteres.
	 */
	onInput(event: Event): void {
		const val = (event.target as HTMLInputElement).value
		this.store.setQuery(val)
	}

	/**
	 * Envía la consulta actual y navega a la página de búsqueda/exploración principal.
	 */
	onSubmit(): void {
		if (this.store.query().trim()) {
			this.store.goToExplore()
		}
	}
}

	<!-- Backdrop -->
	<div
		class="fixed inset-0 z-[100] flex items-start justify-center bg-black/50 px-3 pt-[8vh] backdrop-blur-sm sm:px-4 sm:pt-[10vh]"
		tabindex="-1"
		role="button"
		[attr.aria-label]="'messages.close_search' | translate"
		(click)="store.closePanel()"
		(keydown.enter)="store.closePanel()"
		(keydown.space)="store.closePanel()"
	>
		<!-- Panel -->
		<div
			class="w-full max-w-2xl overflow-hidden rounded-3xl bg-surface-container text-on-surface shadow-5"
			tabindex="-1"
			role="presentation"
			(click)="$event.stopPropagation()"
			(keydown.enter)="$event.stopPropagation()"
			(keydown.space)="$event.stopPropagation()"
			style="animation: searchFadeUp 0.18s ease"
		>
			<!-- Search input row -->
			<div class="flex items-center gap-3 border-b border-outline px-4 py-3 sm:px-5 sm:py-4">
				<i class="pi pi-search text-xl text-on-surface-muted"></i>
				<input
					#searchInput
					type="search"
					class="min-w-0 flex-1 bg-transparent text-body-md font-semibold text-on-surface outline-none placeholder:text-on-surface-muted"
					[placeholder]="t.translate('search.placeholder')"
					autocomplete="off"
					[value]="store.query()"
					(input)="onInput($event)"
					(keydown.enter)="onSubmit()"
				/>
				@if (store.loading()) {
					<span class="size-5 animate-spin rounded-pill border-2 border-outline border-t-primary"></span>
				}
				<kbd class="hidden rounded-lg border border-outline bg-surface px-2 py-1 text-label-md font-semibold text-on-surface-muted lg:block">
					ESC
				</kbd>
			</div>

			<!-- Tab bar (only when there are results) -->
			@if (hasResults()) {
				<div class="flex gap-1 border-b border-outline px-3 pb-0 pt-3 sm:px-4">
					@for (tab of [{ key: 'all', label: t.translate('search.tab.all') }, { key: 'users', label: t.translate('search.tab.users') }, { key: 'posts', label: t.translate('search.tab.recipes') }]; track tab.key) {
						<button
							type="button"
							class="min-h-11 rounded-t-xl px-4 pb-3 pt-2 text-body-sm font-bold transition-colors duration-fast ease-standard"
							[ngClass]="store.tab() === tab.key
								? 'border-b-2 border-primary text-primary'
								: 'text-on-surface-muted hover:text-on-surface'"
							(click)="store.setTab($any(tab.key))"
						>
							{{ tab.label }}
						</button>
					}
				</div>
			}

			<!-- Results -->
			<div class="max-h-[60vh] overflow-y-auto p-3">

				@if (!store.query().trim()) {
					<!-- Empty / hint state -->
					<div class="grid min-h-32 place-items-center text-center">
						<div>
							<i class="pi pi-search text-3xl text-primary"></i>
							<p class="mt-2 text-body-sm font-bold text-on-surface-muted">{{ 'search.hint' | translate }}</p>
							<p class="mt-1 text-label-md font-semibold text-on-surface-muted">{{ 'search.hint_shortcut' | translate }}</p>
						</div>
					</div>

				} @else if (store.loading()) {
					<!-- Skeletons -->
					<div class="grid gap-2">
						@for (i of [1,2,3,4]; track i) {
							<div class="flex items-center gap-3 rounded-2xl p-3">
								<div class="size-11 animate-pulse rounded-pill bg-surface-container-high"></div>
								<div class="flex-1 space-y-2">
									<div class="h-3.5 w-40 animate-pulse rounded-pill bg-surface-container-high"></div>
									<div class="h-3 w-24 animate-pulse rounded-pill bg-surface-container-high"></div>
								</div>
							</div>
						}
					</div>

				} @else if (!hasResults()) {
					<!-- No results -->
					<div class="grid min-h-32 place-items-center text-center">
						<div>
							<i class="pi pi-search text-2xl text-on-surface-muted"></i>
							<p class="mt-2 text-body-sm font-bold text-on-surface-muted">{{ 'search.no_results' | translate }} "{{ store.query() }}"</p>
							<p class="text-label-md font-semibold text-on-surface-muted">{{ 'search.try_another' | translate }}</p>
						</div>
					</div>

				} @else {

					<!-- Users section -->
					@if (showUsers()) {
						<p class="mb-1 px-3 pt-1 text-label-md font-bold text-on-surface-muted">{{ 'search.users_label' | translate }}</p>
						@for (user of store.users(); track user.userId) {
							<button
								type="button"
								class="flex min-h-[64px] w-full items-center gap-3 rounded-2xl p-3 text-left transition-colors duration-fast ease-standard hover:bg-surface-container-high active:scale-[0.98]"
								(click)="store.goToUser(user.username)"
							>
								<span class="grid size-11 shrink-0 place-items-center overflow-hidden rounded-pill bg-primary-container">
									@if (user.photoUrl) {
										<img [src]="user.photoUrl" [alt]="user.username" class="size-full object-cover" />
									} @else {
										<i class="pi pi-user text-primary"></i>
									}
								</span>
								<span class="min-w-0 flex-1">
									<span class="block truncate text-body-sm font-bold text-on-surface">
										{{ user.fullName || user.username }}
									</span>
									<span class="block truncate text-label-md font-semibold normal-case tracking-normal text-on-surface-muted">
										&#64;{{ user.username }}
									</span>
								</span>
								<i class="pi pi-arrow-right text-sm text-primary"></i>
							</button>
						}
					}

					<!-- Posts section -->
					@if (showPosts()) {
						<p class="mb-1 mt-2 px-3 pt-1 text-label-md font-bold text-on-surface-muted">{{ 'search.recipes_label' | translate }}</p>
						@for (post of store.posts(); track post.id) {
							<button
								type="button"
								class="flex min-h-[64px] w-full items-center gap-3 rounded-2xl p-3 text-left transition-colors duration-fast ease-standard hover:bg-surface-container-high active:scale-[0.98]"
								(click)="store.goToPost(post.id)"
							>
								<!-- Thumbnail -->
								<span class="grid size-11 shrink-0 place-items-center overflow-hidden rounded-xl bg-secondary-container">
									@if (post.thumbnailUrl) {
										<img [src]="post.thumbnailUrl" [alt]="post.title || ''" class="size-full object-cover" />
									} @else {
										<i class="pi pi-image text-secondary"></i>
									}
								</span>
								<span class="min-w-0 flex-1">
									<span class="block truncate text-body-sm font-bold text-on-surface">
										{{ post.title || ('search.no_title' | translate) }}
									</span>
									<span class="flex items-center gap-3 text-label-md font-semibold normal-case tracking-normal text-on-surface-muted">
										<span><i class="pi pi-heart mr-1"></i>{{ post.likesCount }}</span>
										<span><i class="pi pi-comment mr-1"></i>{{ post.commentsCount }}</span>
									</span>
								</span>
								<i class="pi pi-arrow-right text-sm text-primary"></i>
							</button>
						}
					}

					<!-- See all in explore -->
					<div class="mt-2 border-t border-outline pt-2">
						<button
							type="button"
							class="flex min-h-11 w-full items-center justify-center gap-2 rounded-2xl px-3 py-3 text-body-sm font-bold text-primary transition-colors duration-fast ease-standard hover:bg-primary-container"
							(click)="store.goToExplore()"
						>
							<i class="pi pi-external-link"></i>
							{{ 'search.view_all' | translate }}
						</button>
					</div>
				}
			</div>
		</div>
	</div>
}
